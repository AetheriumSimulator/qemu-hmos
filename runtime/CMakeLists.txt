cmake_minimum_required(VERSION 3.16)
project(aether_runtime LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(aether_runtime SHARED src/aether_runtime.cpp)
target_compile_features(aether_runtime PRIVATE cxx_std_17)
target_include_directories(aether_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(aether_runtime PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
option(AETHER_RUNTIME_TESTS "Build host contract tests with synthetic cores" OFF)
if(AETHER_RUNTIME_TESTS)
    enable_testing()
    foreach(kind good wrong missing)
        add_library(mock_${kind} SHARED tests/mock_core.cpp)
        target_compile_definitions(mock_${kind} PRIVATE MOCK_${kind}=1)
        target_link_libraries(mock_${kind} PRIVATE Threads::Threads)
    endforeach()
    add_executable(runtime_test tests/runtime_test.cpp)
    target_link_libraries(runtime_test PRIVATE aether_runtime Threads::Threads)
    foreach(case arguments missing wrong once concurrent)
        add_test(NAME runtime_${case} COMMAND runtime_test ${case}
            $<TARGET_FILE:mock_good> $<TARGET_FILE:mock_wrong> $<TARGET_FILE:mock_missing>)
    endforeach()
    add_executable(c_header_test tests/c_header_test.c)
    target_link_libraries(c_header_test PRIVATE aether_runtime)
    add_test(NAME runtime_c_header COMMAND c_header_test)
endif()
option(AETHER_RUNTIME_NAPI "Build the optional HarmonyOS N-API module" OFF)
if(AETHER_RUNTIME_NAPI)
    add_library(aether_runtime_napi SHARED harmonyos/napi_runtime.cpp)
    target_compile_features(aether_runtime_napi PRIVATE cxx_std_17)
    target_link_libraries(aether_runtime_napi PRIVATE aether_runtime ace_napi.z)
endif()
